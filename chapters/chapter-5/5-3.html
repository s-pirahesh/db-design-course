1. قیود جامعیتی صفتی (AC)

تعریف: از محدودیت دامنه‌ای صفت ناشی می‌شوند.

شامل:

  • نوع صفت (Data Type)
  • دامنه مقادیر (Domain)
  • محدودیت مقادیر صفت

مثال 1: نوع داده

CREATE TABLE Student (
    StudentID INT,  -- AC: فقط عدد صحیح
    Name NVARCHAR(50),  -- AC: رشته، حداکثر 50 کاراکتر
    BirthDate DATE,  -- AC: فقط تاریخ
    GPA DECIMAL(4,2),  -- AC: عدد اعشاری، 4 رقم کل، 2 رقم اعشار
    IsActive BIT  -- AC: فقط 0 یا 1
);

نقض AC: 

INSERT INTO Student VALUES ('ABC', 'علی', ...);  -- ❌ StudentID باید INT باشه
INSERT INTO Student VALUES (100, 'علی'*100, ...);  -- ❌ Name بیش از 50 کاراکتر
INSERT INTO Student VALUES (100, 'علی', '2025-13-40', ...);  -- ❌ تاریخ نامعتبر
INSERT INTO Student VALUES (100, 'علی', ..., 100.123, ...);  -- ❌ GPA بیش از 2 رقم اعشار

مثال 2: محدوده مقادیر

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    Age INT CHECK (Age >= 18 AND Age <= 65),  -- AC: سن بین 18 تا 65
    Salary DECIMAL(10,2) CHECK (Salary > 0),  -- AC: حقوق مثبت
    Gender CHAR(1) CHECK (Gender IN ('M', 'F'))  -- AC: فقط M یا F
);

2. قیود جامعیتی دامنه‌ای (DC)

تعریف: مشخص کننده نوع یا مجموعه مقادیر هستند.

در SQL استاندارد به سیستم داده می‌شوند:

  • SQL-Server: CHECK
  • Assertion
  • Trigger

مثال 1: محدوده ساده

CREATE TABLE S (
    S# INT CHECK (S# > 1 AND S# < 100),  -- DC: کد فروشنده بین 1 تا 100
    SName NVARCHAR(50),
    City NVARCHAR(30),
    Status INT
);

تست: 

INSERT INTO S VALUES (0, 'Smith', 'Tehran', 100);  -- ❌ S# کمتر از 1
INSERT INTO S VALUES (50, 'Jones', 'Shiraz', 200);  -- ✅
INSERT INTO S VALUES (150, 'Blake', 'Isfahan', 150);  -- ❌ S# بیشتر از 100

مثال 2: محدوده پیچیده

CREATE TABLE Product (
    ProductID INT PRIMARY KEY,
    ProductName NVARCHAR(100),
    Price DECIMAL(10,2) CHECK (Price >= 1000 AND Price <= 10000000),  -- DC
    DiscountPercent DECIMAL(5,2) CHECK (DiscountPercent >= 0 AND DiscountPercent <= 100),  -- DC
    StockQuantity INT CHECK (StockQuantity >= 0)  -- DC
);

مثال 3: Domain با CONSTRAINT

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    Email VARCHAR(100),
    CONSTRAINT chk_email CHECK (Email LIKE '%@%.%')  -- DC: فرمت ایمیل
);

3. قیود جامعیتی رابطه‌ای (RC)

تعریف: محدودیتی است که فقط ناظر یک رابطه است. گاهی به آن محدودیت درون رابطه‌ای نیز می‌گویند.

مثال 1: محدودیت بین دو صفت

-- S(S#, SName, City, Account)
-- مشتریانی که شهرشان C2 است، Account شان نمی‌تواند بیشتر از 1000 باشد.

CREATE TABLE S (
    S# INT PRIMARY KEY,
    SName NVARCHAR(50),
    City NVARCHAR(30),
    Account DECIMAL(10,2),
    CONSTRAINT chk_city_account 
        CHECK (NOT (City = 'C2' AND Account > 1000))  -- RC
);

تست: 

INSERT INTO S VALUES (10, 'Smith', 'C1', 2000);  -- ✅ شهر C1 → محدودیت نداره
INSERT INTO S VALUES (20, 'Jones', 'C2', 500);  -- ✅ شهر C2، Account < 1000
INSERT INTO S VALUES (30, 'Blake', 'C2', 1500);  -- ❌ شهر C2، Account > 1000

مثال 2: محدودیت تاریخی

CREATE TABLE Project (
    ProjectID INT PRIMARY KEY,
    ProjectName NVARCHAR(100),
    StartDate DATE,
    EndDate DATE,
    CONSTRAINT chk_dates CHECK (EndDate > StartDate)  -- RC
);

تست: 

INSERT INTO Project VALUES (1, 'پروژه A', '2024-01-01', '2024-12-31');  -- ✅
INSERT INTO Project VALUES (2, 'پروژه B', '2024-06-01', '2024-03-01');  -- ❌ EndDate < StartDate

مثال 3: محدودیت پیچیده

CREATE TABLE Loan (
    LoanID INT PRIMARY KEY,
    Amount DECIMAL(15,2),
    InterestRate DECIMAL(5,2),
    Duration INT,  -- به ماه
    CONSTRAINT chk_loan 
        CHECK (
            (Amount < 100000000 AND Duration <= 60) OR
            (Amount >= 100000000 AND InterestRate >= 5)
        )  -- RC: اگر وام کمتر از 100 میلیون، مدت حداکثر 60 ماه
           --     اگر بیشتر از 100 میلیون، نرخ حداقل 5%
);

4. قیود جامعیتی پایگاهی (DBC)

تعریف: محدودیت‌هایی هستند که بین چندین رابطه وجود دارند.

⚠️ نکته: محدودیت‌های عام را نمی‌توان با Trigger به طور همزمان 2 محدودیت اعمال نمود، ولی با Assertion می‌توانیم.

مثال 1: تعداد ثبت‌نام

-- محدودیت: هر دانشجو حداکثر 5 درس می‌تونه بگیره

-- با Trigger:
CREATE TRIGGER trg_max_enrollment
BEFORE INSERT ON Enrollment
FOR EACH ROW
BEGIN
    DECLARE course_count INT;
    SELECT COUNT(*) INTO course_count
    FROM Enrollment
    WHERE StudentID = NEW.StudentID;
    
    IF course_count >= 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'حداکثر 5 درس مجاز است';
    END IF;
END;

مثال 2: موجودی حساب

-- محدودیت: مجموع موجودی حساب‌های یک مشتری نباید منفی باشه

-- با Assertion (SQL استاندارد):
CREATE ASSERTION total_balance_positive
CHECK (
    NOT EXISTS (
        SELECT C.CustomerID
        FROM Customer C
        WHERE (
            SELECT SUM(Balance)
            FROM Account
            WHERE CustomerID = C.CustomerID
        ) < 0
    )
);

مثال 3: ظرفیت کلاس

-- محدودیت: تعداد ثبت‌نام در یک کلاس نباید بیشتر از ظرفیت باشه

CREATE TRIGGER trg_class_capacity
BEFORE INSERT ON Enrollment
FOR EACH ROW
BEGIN
    DECLARE current_count INT;
    DECLARE max_capacity INT;
    
    SELECT COUNT(*) INTO current_count
    FROM Enrollment
    WHERE CourseID = NEW.CourseID AND Semester = NEW.Semester;
    
    SELECT Capacity INTO max_capacity
    FROM Course
    WHERE CourseID = NEW.CourseID;
    
    IF current_count >= max_capacity THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'ظرفیت کلاس پر است';
    END IF;
END;

ساختار Trigger

CREATE TRIGGER trigger_name
{BEFORE | AFTER | INSTEAD OF} {INSERT | UPDATE | DELETE}
ON table_name
FOR EACH ROW
BEGIN
    -- Custom Logic
END;

مثال 1: Audit Log

-- Log کردن تغییرات حقوق
CREATE TABLE SalaryLog (
    LogID INT PRIMARY KEY AUTO_INCREMENT,
    EmpID INT,
    OldSalary DECIMAL(10,2),
    NewSalary DECIMAL(10,2),
    ChangeDate DATETIME,
    ChangedBy VARCHAR(50)
);

CREATE TRIGGER trg_salary_change
AFTER UPDATE ON Employee
FOR EACH ROW
BEGIN
    IF OLD.Salary <> NEW.Salary THEN
        INSERT INTO SalaryLog 
        VALUES (NULL, NEW.EmpID, OLD.Salary, NEW.Salary, NOW(), USER());
    END IF;
END;

مثال 2: Validation پیچیده

-- اعتبارسنجی قبل از درج
CREATE TRIGGER trg_validate_student
BEFORE INSERT ON Student
FOR EACH ROW
BEGIN
    -- بررسی سن
    IF YEAR(CURDATE()) - YEAR(NEW.BirthDate) < 15 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'سن دانشجو باید حداقل 15 سال باشد';
    END IF;
    
    -- بررسی کد ملی (10 رقمی)
    IF LENGTH(NEW.NationalID) <> 10 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'کد ملی باید 10 رقمی باشد';
    END IF;
END;

مثال 3: Cascade دستی

-- Cascade دستی برای Delete
CREATE TRIGGER trg_delete_student
BEFORE DELETE ON Student
FOR EACH ROW
BEGIN
    -- حذف ثبت‌نام‌ها
    DELETE FROM Enrollment WHERE StudentID = OLD.StudentID;
    
    -- حذف نمرات
    DELETE FROM Grades WHERE StudentID = OLD.StudentID;
    
    -- Log
    INSERT INTO DeleteLog VALUES (OLD.StudentID, NOW());
END;

2. Assertion (اظهاریه)

تعریف: می‌توانیم روی DBMS قواعد جامعیتی را با آن کنترل نماییم.

ساختار: 

CREATE ASSERTION assertion_name
CHECK (
    NOT EXISTS (query)
);

مثال 1: محدودیت تعداد

-- هر استاد حداکثر 3 درس تدریس کنه
CREATE ASSERTION max_courses_per_instructor
CHECK (
    NOT EXISTS (
        SELECT InstructorID
        FROM Teaching
        GROUP BY InstructorID
        HAVING COUNT(*) > 3
    )
);

مثال 2: محدودیت مجموع

-- مجموع حقوق یک اداره نباید بیشتر از بودجه‌اش باشه
CREATE ASSERTION department_budget
CHECK (
    NOT EXISTS (
        SELECT D.DeptID
        FROM Department D
        WHERE (
            SELECT SUM(Salary)
            FROM Employee E
            WHERE E.DeptID = D.DeptID
        ) > D.Budget
    )
);

مثال 3: محدودیت منطقی

-- هر دانشجو باید حداقل 12 واحد بگیره
CREATE ASSERTION min_units
CHECK (
    NOT EXISTS (
        SELECT StudentID
        FROM Enrollment
        GROUP BY StudentID, Semester
        HAVING SUM(Credits) < 12
    )
);

✍️ تمرینات بخش ۵.۳

تمرین 1 (تشریحی): بانک

طراحی سیستم بانکی با قیود زیر: 

موجودیت‌ها:
- Customer (CustomerID, Name, Phone, Address)
- Account (AccountID, CustomerID, Balance, AccountType)
- Transaction (TxnID, AccountID, Amount, TxnDate, TxnType)

قیود جامعیتی:
1. Balance نباید منفی باشه (DC)
2. AccountType فقط 'Savings' یا 'Current' (DC)
3. مجموع موجودی حساب‌های یک مشتری حداقل 500,000 (DBC)
4. تعداد تراکنش‌های روزانه هر حساب حداکثر 10 (RC)
5. Amount تراکنش بین 10,000 تا 50,000,000 (DC)

انجام دهید:

  • الف) SQL کامل با همه قیود
  • ب) 3 Trigger برای قیود پیچیده
  • ج) Test Case‌های نقض و رعایت
  • د) Assertion برای قید DBC

تمرین 2 (تستی):

کدام گزینه درست است؟ 

الف) محدودیت‌های عام را می‌توان با Trigger به طور همزمان 2 محدودیت اعمال نمود.
ب) اگر کلید اصلی جزو کلید خارجی باشد، NULLIFY نمودن صحیح است.
ج) CASCADE باعث می‌شود عملیات روی والد، روی فرزند هم اعمال شود.
د) RC محدودیتی است که بین چندین رابطه وجود دارد.

تمرین 3 (تحلیلی): فروشگاه

جداول زیر را بررسی کنید: 

CREATE TABLE Product (
    ProductID INT PRIMARY KEY,
    ProductName NVARCHAR(100),
    Price DECIMAL(10,2),
    StockQuantity INT
);

CREATE TABLE OrderItem (
    OrderID INT,
    ProductID INT,
    Quantity INT,
    PRIMARY KEY (OrderID, ProductID),
    FOREIGN KEY (ProductID) REFERENCES Product(ProductID)
);

سؤالات:

  • الف) چه قیودی اضافه کنیم؟ (حداقل 5 قید)
  • ب) Trigger برای کاهش StockQuantity بعد از فروش
  • ج) چک کردن موجودی کافی قبل از فروش
  • د) Assertion برای مجموع فروش روزانه

تمرین 4 (طراحی): دانشگاه

سیستم ثبت‌نام با قیود پیچیده: 

قیود:
1. هر دانشجو حداقل 12 و حداکثر 20 واحد
2. پیش‌نیازها رعایت بشه
3. ظرفیت کلاس رعایت بشه
4. تداخل زمانی نباشه
5. معدل حداقل 12 برای ادامه تحصیل
6. حذف و اضافه تا 2 هفته اول

طراحی کنید:

  • الف) جداول با قیود
  • ب) 5 Trigger
  • ج) 3 Assertion
  • د) Test Scenario کامل

تمرین 5 (مفهومی):

سؤال 1: چرا DBMS‌ها کند هستند؟ 3 دلیل با مثال.

سؤال 2: تفاوت CASCADE و RESTRICT را با سناریو کامل توضیح دهید.

سؤال 3: چه موقع از Trigger استفاده کنیم و چه موقع از Assertion؟

تمرین 6 (پیشرفته): سیستم پیچیده

سیستم مدیریت پروژه:
- Employee (چند مهارت، چند پروژه)
- Project (بودجه، مدت زمان، اعضا)
- Assignment (تخصیص کارمند به پروژه)
- Skill (مهارت‌های لازم برای پروژه)

قیود پیچیده:
1. مجموع حقوق اعضای پروژه <= بودجه
2. هر کارمند حداکثر 3 پروژه همزمان
3. مهارت‌های کارمند با نیاز پروژه مطابقت داشته باشه
4. تاریخ‌های پروژه‌ها تداخل نداشته باشن
5. حداقل یک Project Manager در هر پروژه

پیاده‌سازی:

  • الف) طراحی کامل
  • ب) همه قیود (AC, DC, RC, DBC)
  • ج) 10 Trigger
  • د) 5 Assertion

📌 نکات کلیدی فصل 5

✅ جامعیت:

  • صحت + دقت + سازگاری
  • در تمام عمر پایگاه داده
  • مسئول: DBMS

✅ قیود عام:

  • C1: Entity Integrity (PK NOT NULL)
  • C2: Referential Integrity (FK → PK)

✅ قیود خاص:

  • AC: صفتی (نوع، دامنه)
  • DC: دامنه‌ای (CHECK)
  • RC: رابطه‌ای (یک جدول)
  • DBC: پایگاهی (چند جدول)

✅ مکانیزم‌ها:

  • Trigger: قطعه برنامه ذخیره شده
  • Assertion: Query چک‌کننده

✅ ON DELETE/UPDATE:

  • CASCADE: آبشاری
  • RESTRICT: منع
  • NO ACTION: Trigger
  • NULLIFY: NULL کن
  • SET DEFAULT: مقدار پیش‌فرض

✅ تفاوت Integrity و Security:

  • Integrity: صحت داده
  • Security: محافظت از داده
