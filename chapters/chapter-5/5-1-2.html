قیود عام (MC - Mandatory Constraints): ناوابسته به داده‌ها، در تمامی DBMS‌ها برقرار هستند.

C1: جامعیت موجودیتی (Entity Integrity)

تعریف

قاعده C1 ناظر به کلید اصلی است:

دلیل: کلید اصلی عامل تشخیص موجودیت‌هاست و این کلید نمی‌تواند مقدار ناشناخته داشته باشد.

مثال - Entity Integrity

❌ نقض C1: 

CREATE TABLE Student (
    StudentID INT PRIMARY KEY,
    Name NVARCHAR(50),
    Major NVARCHAR(30)
);

-- تلاش برای درج با PK=NULL
INSERT INTO Student VALUES (NULL, 'علی', 'CS');
-- ❌ خطا! Cannot insert NULL into PRIMARY KEY

-- تلاش با PK ترکیبی
CREATE TABLE Enrollment (
    StudentID INT,
    CourseID INT,
    Grade DECIMAL(4,2),
    PRIMARY KEY (StudentID, CourseID)
);

INSERT INTO Enrollment VALUES (100, NULL, 18.5);
-- ❌ خطا! CourseID جزو PK است و نمی‌تونه NULL باشه

✅ رعایت C1: 

-- همه PK‌ها مقدار دارن
INSERT INTO Student VALUES (100, 'علی', 'CS');  ✅
INSERT INTO Student VALUES (110, 'سارا', 'Math');  ✅

INSERT INTO Enrollment VALUES (100, 201, 18.5);  ✅
INSERT INTO Enrollment VALUES (110, 202, 17.0);  ✅

اعمال C1

C1 را چگونه می‌توان اعمال نمود؟ 

-- روش 1: با PRIMARY KEY (توصیه شده ✅)
CREATE TABLE Student (
    StudentID INT PRIMARY KEY,  -- خودکار NOT NULL
    Name NVARCHAR(50)
);

-- روش 2: با NOT NULL صریح
CREATE TABLE Student (
    StudentID INT NOT NULL,
    Name NVARCHAR(50),
    PRIMARY KEY (StudentID)
);

-- روش 3: PK ترکیبی
CREATE TABLE Enrollment (
    StudentID INT NOT NULL,  -- صریح (اختیاری)
    CourseID INT NOT NULL,   -- صریح (اختیاری)
    Grade DECIMAL(4,2),
    PRIMARY KEY (StudentID, CourseID)  -- خودکار NOT NULL برای هر دو
);

نکته: با تعریف PRIMARY KEY، DBMS خودکار NOT NULL رو اعمال می‌کنه! ✅

مثال - Referential Integrity

طراحی: 

-- رابطه R2 (ارجاع شونده)
CREATE TABLE Department (
    DeptID INT PRIMARY KEY,
    DeptName NVARCHAR(50)
);

-- رابطه R1 (ارجاع کننده)
CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    DeptID INT,  -- FK
    FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
);

داده نمونه: 

Department:
+----+--------+
| ID | Name   |
+----+--------+
| 10 | CS     |
| 20 | Math   |
+----+--------+

Employee:
+-----+--------+--------+
| EID | Name   | DeptID |
+-----+--------+--------+
| 100 | علی    | 10     |  ✅ اشاره به 10 (موجود)
| 110 | سارا   | 20     |  ✅ اشاره به 20 (موجود)
+-----+--------+--------+

❌ نقض C2: 

-- تلاش برای درج با DeptID غیرموجود
INSERT INTO Employee VALUES (120, 'رضا', 99);
-- ❌ خطا! DeptID=99 در Department وجود نداره

-- تلاش برای UPDATE به مقدار غیرموجود
UPDATE Employee SET DeptID = 99 WHERE EmpID = 100;
-- ❌ خطا! نقض Referential Integrity

✅ رعایت C2: 

-- ابتدا اداره رو اضافه کن
INSERT INTO Department VALUES (30, 'Physics');

-- حالا می‌تونی کارمند با DeptID=30 اضافه کنی
INSERT INTO Employee VALUES (120, 'رضا', 30);  ✅

CASCADE (آبشاری)

عملیات روی رکورد والد، روی رکوردهای فرزند هم اعمال می‌شود. 

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    DeptID INT,
    FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
        ON DELETE CASCADE  -- حذف آبشاری
        ON UPDATE CASCADE  -- بروزرسانی آبشاری
);

مثال DELETE CASCADE:

| 10 | CS     |  ← می‌خوایم حذف کنیم

| 110 | سارا   | 10     |  ← وابسته به Dept=10

RESTRICT (محدود کردن)

اگر رکورد فرزندی وجود داشته باشد، از حذف یا بروزرسانی والد جلوگیری می‌نماید. 

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    DeptID INT,
    FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
        ON DELETE RESTRICT  -- منع حذف
        ON UPDATE RESTRICT  -- منع بروزرسانی
);

مثال: 

DELETE FROM Department WHERE DeptID = 10;

→ RESTRICT: ❌ خطا! چون کارمندانی با DeptID=10 وجود دارن

راه‌حل:
1. ابتدا کارمندها رو حذف کن
2. سپس اداره رو حذف کن
NO ACTION (عدم عمل)

Trigger‌ها را در نظر می‌گیرد.

مشابه RESTRICT ولی Trigger‌ها اجرا می‌شن. 

FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
NULLIFY (Null کردن)

اگر رکورد مربوطه حذف گردد، آنرا NULL کن. 

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    DeptID INT,  -- می‌تونه NULL باشه
    FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
        ON DELETE SET NULL  -- Null کن
        ON UPDATE SET NULL
);

مثال: 

DELETE FROM Department WHERE DeptID = 10;

→ NULLIFY: DeptID کارمندان 100 و 110 می‌شه NULL! ✅

Employee:
+-----+--------+--------+
| 100 | علی    | NULL   |  ← Null شد
| 110 | سارا   | NULL   |  ← Null شد
| 120 | رضا    | 20     |
+-----+--------+--------+

⚠️ نکته مهم: اگر کلید اصلی جزو کلید خارجی باشد، NULLIFY نمودن اشتباه است! 

-- ❌ اشتباه!
CREATE TABLE Enrollment (
    StudentID INT,
    CourseID INT,
    Grade DECIMAL(4,2),
    PRIMARY KEY (StudentID, CourseID),
    FOREIGN KEY (StudentID) REFERENCES Student(StudentID)
        ON DELETE SET NULL  -- ❌ نمی‌شه! StudentID جزو PK است
);
SET DEFAULT (مقدار پیش‌فرض)

اگر رکورد حذف شد، مقدار پیش‌فرض را قرار بده. 

CREATE TABLE Employee (
    EmpID INT PRIMARY KEY,
    Name NVARCHAR(50),
    DeptID INT DEFAULT 99,  -- اداره "نامشخص"
    FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
        ON DELETE SET DEFAULT
);

⚠️ نکته: مقدار پیش‌فرض حتماً باید در جدول ارجاع شونده وجود داشته باشد! 

-- ابتدا اداره "نامشخص" رو اضافه کن
INSERT INTO Department VALUES (99, 'نامشخص');

-- حالا DELETE با SET DEFAULT کار می‌کنه
DELETE FROM Department WHERE DeptID = 10;

مثال کامل C2

-- جداول
CREATE TABLE S (
    S# INT PRIMARY KEY,
    SName NVARCHAR(50),
    City NVARCHAR(30),
    Status INT
);

CREATE TABLE P (
    P# INT PRIMARY KEY,
    PName NVARCHAR(50),
    Color NVARCHAR(20),
    Weight INT
);

CREATE TABLE SP (
    S# INT,
    P# INT,
    Qty INT,
    PRIMARY KEY (S#, P#),
    FOREIGN KEY (S#) REFERENCES S(S#)
        ON DELETE NO ACTION
        ON UPDATE CASCADE,
    FOREIGN KEY (P#) REFERENCES P(P#)
        ON DELETE CASCADE
        ON UPDATE RESTRICT
);

تحلیل:

برای S#:

  • ON DELETE NO ACTION: اگر فروشنده‌ای رو حذف کنیم که تولید داره → ❌ خطا
  • ON UPDATE CASCADE: اگر S# فروشنده عوض شه → در SP هم عوض می‌شه

برای P#:

  • ON DELETE CASCADE: اگر قطعه‌ای رو حذف کنیم → رکوردهای مربوط در SP هم حذف می‌شن
  • ON UPDATE RESTRICT: اگر P# قطعه عوض شه و تولید داره → ❌ خطا
